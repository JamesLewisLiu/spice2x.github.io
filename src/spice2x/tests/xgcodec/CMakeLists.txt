cmake_minimum_required(VERSION 3.16)
project(xg_codec_tests LANGUAGES CXX)
enable_testing()
add_executable(xg_codec_tests codec_test.cpp)
target_compile_features(xg_codec_tests PRIVATE cxx_std_20)
target_compile_definitions(xg_codec_tests PRIVATE FMT_HEADER_ONLY WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(xg_codec_tests PRIVATE stubs ../..)
target_link_libraries(xg_codec_tests PRIVATE ole32 uuid advapi32)
if(MINGW)
    target_link_options(xg_codec_tests PRIVATE -static)
endif()
add_test(NAME xg_codec_registry COMMAND xg_codec_tests)
set(XG_CODEC_AX "" CACHE FILEPATH "Optional 32-bit CLVSD.ax for the real COM integration test")
if(XG_CODEC_AX)
    add_test(NAME xg_codec_real_filter COMMAND xg_codec_tests "${XG_CODEC_AX}")
endif()
